f_floorTick(float price) =>

f_shape(int offset, float atr, float support, float meanVolume, float minRVOL) =>
    float span = high[offset] - low[offset]
    bool volumeOK = not na(volume[offset]) and volume[offset] > 0 and not na(meanVolume) and meanVolume > 0
    bool result = (volumeOK and not na(atr) and atr > 0 and span > 0 and not na(close[offset + 1])
         and volume[offset] / meanVolume >= minRVOL
         and math.abs(close[offset] - open[offset]) <= atr * maxEvidenceBody
         and span <= atr * maxEvidenceRange
         and (close[offset] - low[offset]) / span >= evidenceCloseLocation
         and (math.min(open[offset], close[offset]) - low[offset]) / span >= minimumLowerWick
         and math.max(close[offset + 1] - close[offset], 0.0) <= atr * maxDownsideProgress
         and not na(support) and low[offset] <= support + atr * supportDistanceATR and close[offset] >= support)
    result

// All history-dependent TA calculations are executed globally on every bar.
float atrSeries = ta.atr(atrLength)
float priorATR = atrSeries[1]
float volumeMean = ta.sma(volume, volumeLength)
float validVolumeBars = math.sum(not na(volume) and volume >= 0 ? 1.0 : 0.0, volumeLength)
float baseValidVolumeBars = math.sum(not na(volume) and volume >= 0 ? 1.0 : 0.0, evidenceWindow)
float supportSeries = ta.lowest(low, supportLength)
float priorVolumeMean = volumeMean[1]
bool priorVolumeReady = not na(validVolumeBars[1]) and validVolumeBars[1] >= volumeLength and not na(priorVolumeMean) and priorVolumeMean > 0
// Source metrics deliberately use CURRENT-inclusive ATR and volume average.
// Equivalent highest/lowest expressions are evaluated globally each bar.
float sourceATR = ta.atr(sourceATRLength)
float sourceVolMA = ta.sma(volume, sourceVolumeLength)
float sourceRVOL = sourceVolMA > 0 ? volume / sourceVolMA : 0.0
float sourceRange = math.max(high - low, syminfo.mintick)
float sourceCLV = (close - low) / sourceRange
float sourcePriorLow = ta.lowest(low[1], sourceSweepLength)
float sourceMicroHigh = ta.highest(high[1], sourceConfirmBOSLength)
bool sourceInitializationReady = not na(sourceATR) and not na(sourceVolMA) and sourceVolMA > 0
float candleRange = high - low
float bodyFraction = candleRange > 0 ? math.abs(close - open) / candleRange : 0.0
float closeLocation = candleRange > 0 ? (close - low) / candleRange : 0.0
bool strongDisplacement = (not na(priorATR) and priorATR > 0 and close > open
     and candleRange >= minimumDisplacementRange * priorATR and bodyFraction >= minimumDisplacementBody
     and closeLocation >= displacementCloseLocation)

f_precisionBounds(float o, float h, float l, float c) =>
    float top = switch precisionZoneMethod
        "Open to low" => o
        "Body to low" => math.max(o, c)
        "Body only" => math.max(o, c)
        "Lower half of candle" => l + (h - l) * 0.50
        => h
    float bottom = precisionZoneMethod == "Body only" ? math.min(o, c) : l
    [top, bottom]

f_sourceDepth(string mode) =>
    switch mode
        "Proximal" => 0.0
        "50%" => 0.50
        "62%" => 0.62
        "70.5%" => 0.705
        "79%" => 0.79
        "Distal + 1 tick" => 1.0

f_sourceBuffer(float width, float atr) =>
    float tickBuffer = sourceFrontRunTicks * syminfo.mintick
    float atrBuffer = sourceFrontRunATR * atr
    switch sourceFrontRunMode
        "ATR" => atrBuffer
        "Ticks" => tickBuffer
        "Off" => 0.0
        => math.min(math.max(tickBuffer, atrBuffer), width * sourceMaxZoneBuffer)

f_sourceInitialReference(float top, float bottom, float atr) =>
    float width = top - bottom
    float base = top - width * f_sourceDepth(sourceEntryMode)
    float value = base + f_sourceBuffer(width, atr) + sourceEntryOffsetTicks * syminfo.mintick
    sourceEntryMode == "Distal + 1 tick" ? bottom + syminfo.mintick : value

f_sourceState(int state) =>
    switch state
        0 => "Source ready (not a claim of untouched history)"
        1 => "Source tapped / pending"
        2 => "Source defence condition observed"
        => "Source invalid / touch limit exceeded"

f_sourceColor(FootprintOB z) =>
    z.sourceState == 1 ? sourceTapColor : z.sourceState == 2 ? sourceDefenceColor : z.sourceState < 0 ? inactiveColor : zoneColor

f_saveSourceMarker(int zoneId, label id) =>
    SourceMarker m = SourceMarker.new()
    m.zoneId := zoneId
    m.drawing := id

// Reconstruct observed history since this fixed origin. The source is NOT
// replaced if a later candle looks nicer. Contact is not a broker fill.
f_legState(int originBar, float top, float bottom, float invalidation, float atr) =>
    int contacts = 0
    bool armed = false
    bool broken = false
    int count = bar_index - originBar
    if count > 0
        for step = 1 to count
            int k = count - step
            float failurePrice = invalidationMode == "Close" ? close[k] : low[k]
            if failurePrice < invalidation
                broken := true
            bool overlap = low[k] <= top and high[k] >= bottom
            if armed and overlap
                contacts += 1
                armed := false
            else if close[k] > top + atr * contactDepartureATR and (contacts == 0 or not overlap)
                armed := true
    [contacts, armed, broken]


// SOURCE-COMPATIBLE TAP STATE MACHINE. The order of operations below is the
// attachment's order: departure -> approach -> TAP/adjustment -> defence ->
// timeout -> invalidity. It is intentionally NOT the v8 episode algorithm.
f_sourceTapStep(FootprintOB z) =>
    int state = z.sourceState
    int tapCount = z.sourceTaps
    int tapBar = z.sourceTapBar
    bool hasDeparted = z.sourceDeparted
    int age = bar_index - z.bornBar
    float top = z.top
    float bot = z.bottom
    float entry = z.sourceReference
    float stop = z.invalidation
    bool anyApproach = false
    bool anyTap = false
    bool anyConfirm = false
    bool anyDead = false

    if state >= 0 and not hasDeparted and high >= top + sourceATR * sourceRequireDeparture
        hasDeparted := true
        z.sourceDeparted := true

    bool approaching = state >= 0 and hasDeparted and age >= sourceMinAge and close > entry and low <= entry + sourceATR * sourceApproachATR
    if approaching
        anyApproach := true

    // Exact reference-based source condition: high >= BOT, not high >= ENTRY.
    // It may qualify above the drawn OB or on a bar below the reference.
    bool touched = state >= 0 and hasDeparted and age >= sourceMinAge and low <= entry and high >= bot
    bool swept = low < sourcePriorLow
    bool qualifiedTap = touched and (not sourceRequireSweep or swept)
    if qualifiedTap and (tapBar < 0 or bar_index > tapBar)
        tapCount += 1
        tapBar := bar_index
        state := 1
        z.sourceTaps := tapCount
        z.sourceTapBar := tapBar
        z.sourceState := state

        // Faithfully immediate on TAP 1, NOT after defence confirmation.
        if sourceRaiseAfterFirstTap and tapCount == 1
            float adaptiveReference = math.max(entry, low + sourceATR * sourceRepeatTapATR)
            z.sourceReference := adaptiveReference
            z.sourceAdjustmentBar := bar_index
            z.sourceAdjustmentAt := time_close
            if not na(z.sourceReferenceLine)
                line.set_y1(z.sourceReferenceLine, adaptiveReference)
                line.set_y2(z.sourceReferenceLine, adaptiveReference)
            entry := adaptiveReference

        f_sourceAppearance(z, sourceTapColor)
        anyTap := true
        if sourceShowLabels
            label id = label.new(bar_index, low, "TAP " + str.tostring(tapCount) + "\nNext " + str.tostring(entry, format.mintick),
                 style=label.style_label_up, color=sourceTapColor, textcolor=color.black, size=size.tiny, tooltip=f_sourceTip(z))
            f_saveSourceMarker(z.id, id)

    bool pending = state == 1 and tapBar >= 0 and bar_index - tapBar <= sourceConfirmBars
    bool microBOS = close > sourceMicroHigh
    bool defence = barstate.isconfirmed and pending and close > open and sourceCLV >= sourceConfirmCLV and sourceRVOL >= sourceConfirmRVOL and close > top and microBOS
    if defence
        state := 2
        z.sourceState := state
        f_sourceAppearance(z, sourceDefenceColor)
        anyConfirm := true
        if sourceShowLabels
            label id = label.new(bar_index, low, "DEFENCE CONFIRMED", style=label.style_label_up,
                 color=sourceDefenceColor, textcolor=color.white, size=size.small, tooltip=f_sourceTip(z))
            f_saveSourceMarker(z.id, id)

    // Original source reset: tap count/reference/departure latch are NOT reset.
    if state == 1 and bar_index - tapBar > sourceConfirmBars
        state := 0
        z.sourceState := state
        f_sourceAppearance(z, zoneColor)

    // Exact source > limit, AFTER TAP/defence. At maxTouches=4, TAP 5 can be
    // printed before retirement on this same bar. Do not silently change to >=.
    bool invalid = state >= 0 and (close < stop or tapCount > sourceMaxTouches)
    if invalid
        state := -1
        z.sourceState := state
        f_retireZone(z, close < stop ? "Source live-close stop condition" : "Source tap count exceeded maximum")
        anyDead := true

    z.sourceState := state
    z.active := state >= 0
    [anyApproach, anyTap, anyConfirm, anyDead]

f_registerSSL(array<SSLPool> pools, int scope, float price, int sourceBar, int newId, int rangeKey, float creationATR) =>
    bool created = false
    bool duplicate = false
    int matchIndex = -1
    float bestDistance = na
    if array.size(pools) > 0
        for i = 0 to array.size(pools) - 1
            SSLPool old = array.get(pools, i)
            if old.scope == scope and (old.firstOrigin == sourceBar or old.lastOrigin == sourceBar)
                duplicate := true
            // A CONSUMED old pool cannot qualify a new "untaken" EQL pool.
            if sslShowEqual and old.active and old.scope == scope
                float newWidth = math.max(old.upper, price) - math.min(old.lower, price)
                float anchorDistance = math.abs(price - old.anchor)
                if anchorDistance <= old.tolerance + syminfo.mintick * 0.000001 and newWidth <= old.tolerance + syminfo.mintick * 0.000001
                    if matchIndex == -1 or anchorDistance < bestDistance
                        matchIndex := i
                        bestDistance := anchorDistance
    if not duplicate and not na(price) and price > 0
        SSLPool p = SSLPool.new()
        p.id := newId
        p.scope := scope
        p.firstOrigin := sourceBar
        p.lastOrigin := sourceBar
        p.bornBar := bar_index
        p.knownAt := time_close
        p.rangeKey := rangeKey
        p.lower := price
        p.upper := price
        p.anchor := price
        p.tolerance := math.max(syminfo.mintick, creationATR * sslEqualATR)
        if matchIndex >= 0
            SSLPool old = array.get(pools, matchIndex)
            p.firstOrigin := old.firstOrigin
            p.lower := math.min(old.lower, price)
            p.upper := math.max(old.upper, price)
            p.anchor := old.anchor
            p.tolerance := old.tolerance
            p.members := old.members + 1
            p.state := old.state
            // New composite band starts NOW. Do not stretch the old historical
            // boundary downward/upward or restore its touched/partial state.
            f_finishSSL(old, L_CLUSTERED)
        f_drawSSL(p)
        array.push(pools, p)
        created := true
    created

        float width = evidenceTop - evidenceBottom
        float invalidation = f_floorTick(evidenceBottom - math.max(evidenceATR * invalidationATR, invalidationTicks * syminfo.mintick))
        bool valid = width >= minimumWidthTicks * syminfo.mintick - syminfo.mintick * 0.000001 and width <= maxBaseWidthATR * evidenceATR and invalidation > 0
        bool duplicate = false
        if array.size(setups) > 0
            for i = 0 to array.size(setups) - 1
                FootprintSetup old = array.get(setups, i)
                float shared = width > 0 ? math.max(0.0, math.min(evidenceTop, old.top) - math.max(evidenceBottom, old.bottom)) / width : 0.0
                if old.active and shared >= duplicateOverlap
                    duplicate := true
                // Do not recycle already-used observations into a new cluster.
                if old.used and evidenceStart <= old.knownBar and bar_index - old.knownBar <= evidenceWindow + originPadding
                    duplicate := true
        if duplicate
            evidenceDuplicates += 1
        else if valid
            FootprintSetup f = FootprintSetup.new()
            f.id := nextSetupId
            f.startBar := evidenceStart
            f.knownBar := bar_index
            f.knownAt := time_close
            f.top := evidenceTop
            f.bottom := evidenceBottom
            f.atr := evidenceATR
            f.invalidation := invalidation
            f.rule := rule
            f.evidenceRVOL := evidenceRatio
            f.observations := observations
            if not na(knownStructureHigh) and not structureConsumed
                f.structure := knownStructureHigh
                f.structureOrigin := knownStructureOrigin
                f.structureKnownBar := knownStructureBar
                f.structureKnownAt := knownStructureAt
            array.push(setups, f)
            footprintsObserved += 1
            nextSetupId += 1
            lastDecision := "Footprint #" + str.tostring(f.id) + " observed internally; NOT drawn until the whole sequence confirms"

    while array.size(setups) > maxPendingSetups
        int removeIndex = 0
        bool found = false
        for i = 0 to array.size(setups) - 1
            FootprintSetup f = array.get(setups, i)
            if not f.active and not found
                removeIndex := i
                found := true
        FootprintSetup removed = array.remove(setups, removeIndex)
        if removed.active
            f_stopSetup(removed, "Internal record cap")
    while array.size(zones) > maxZones
        int removeIndex = 0
        bool found = false
        for i = 0 to array.size(zones) - 1
            FootprintOB z = array.get(zones, i)
            if not z.active and not found
                removeIndex := i
                found := true
        FootprintOB removed = array.remove(zones, removeIndex)
        if removed.active
            f_retireZone(removed, "Record cap, no longer tracked")
        f_deleteZone(removed)
